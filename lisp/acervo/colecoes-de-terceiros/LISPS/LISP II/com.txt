;;;==================== load-time error checking ========================

  (defun ai_abort (app msg)
     (defun *error* (s)
        (if old_error (setq *error* old_error))
        (princ)
     )
     (if msg
       (alert (strcat " Application error: "
                      app
                      " \n\n  "
                      msg
                      "  \n"
              )
       )
     )
     (exit)
  )

;;; and then try to load it.
;;;
;;; If it can't be found or it can't be loaded, then abort the
;;; loading of this file immediately, preserving the (autoload)
;;; stub function.

  (cond
     (  (and ai_dcl (listp ai_dcl)))          ; it's already loaded.

     (  (not (findfile "ai_utils.lsp"))                     ; find it

                  (strcat "Can't locate file AI_UTILS.LSP."
                          "\n Check support directory.")))

     (  (eq "failed" (load "ai_utils" "failed"))            ; load it
        (ai_abort "COM" "Can't load file AI_UTILS.LSP"))
  )

  (if (not (ai_acadapp))               ; defined in AI_UTILS.LSP
      (ai_abort "COM" nil)         ; a Nil <msg> supresses
  )                                    ; ai_abort's alert box dialog.

;; Calculator key function.
(defun com_key (key)
   (cond 

      ;; Numeral "0" key
      ((and (= key "0") (not lock))
       (setq num (chr 48))
      )

      ;; Numeral "1" key
      ((and (= key "1") (not lock))
       (setq num (chr 49))
      )

      )

      ;; Numeral "3" key
      ((and (= key "3") (not lock))

      )

      ((and (= key "9") (not lock))
       (setq num (chr 57))
      )

      ((and (= key "x") (not lock))
       (fract_test) 
       (clear_var (list 'expo 'power 'neg 'ctr))
       (setq num "*")
      )

      ;; Addition function command key
      ((and (= key "+") (not lock))
       (fract_test) 
       (clear_var (list 'expo 'power 'neg 'ctr))
       (setq num "+")
      )

      ((and (= key "=") (not lock))

       (setq num "=")
      )

      ;; Division function command key
      ((and (= key "/") (not lock))

      ;; Subtraction function command key
      ((and (= key "-") (not lock))

      ((= key "AC") (com_clear))

;;Input analyzer
(defun read_input ()
   (cond
     (input 
        (if (= (type (read input)) 'INT)
            (setq lenv (strlen input))
        )
        (if (= (type (read input)) 'REAL)
            (setq lenv (- (strlen input) 1))
        )
     ) 
     (t (setq lenv 0))
   )
   (if num 
      (cond
         (fract (com_fraction))
         (expo (com_exponent))
         (t (if (or (numberp (read num)) (= num "dot"))
                (if (<= lenv (- max_digit 2)) 
                    (progn
                      (setq tstn t)
                      (if (not lock) (com_intrpt))
                    )
                )
                (cond 
                    ((= num "error") (print_error))
                    (t (if (not tstn) (setq input "0.0"))
                       (if (not lock) (com_oper))
                    )
                )
            )
         )
      )
   )
)

;; Numeric input interpreter.
(defun com_intrpt () 
    (if (not input) 
        (if (and (/= num (chr 48)) (= num "dot"))
            (setq input (strcat (chr 48) (chr 46)))
            (setq input num)
        )
        (if (= num "dot")
            (if (= (type (read input)) 'INT)
                (setq input (strcat input (chr 46)))
            )
            (cond
              ((and (zerop (read num)) (zerop (read input)) 
                    (= (type (read input)) 'INT)
               )
               (setq input (chr 48))
              )
              (t (if (and (zerop (read input)) 
                          (= (type (read input)) 'INT)
                     )
                     (setq input num)
                     (setq input (strcat input num))
                  )
               )
            )
        )
    )
    (if (= (type (read input)) 'INT)
        (cond
          (funct (print_value (strcat input (chr 46)) rmv))
          (t (print_value (strcat input (chr 46)) nil))
        )
    )
    (if (= (type (read input)) 'REAL)
        (cond
          (funct (print_value input rmv))
          (t (print_value input nil))
        )
    )
    (setq rdc nil)
    (setq saver nil);; always set to nil everytime a numeric key is hit.
    (if input (setq temp input))
    (if funct (setq input nil saver dmv funct nil))
)

;; Calc-exponent function.
(defun com_exponent ()
    (cond
      (expnt (if (and num (/= num "dot"))
         (setq expnt (strcat expnt num)))
      )
      (t (setq expnt num))
    )
    (setq lenx (strlen expnt))
    (if (> lenx 2) (setq expnt (substr expnt (- lenx 1))))
    (if neg (setq expnt (strcat "-" expnt))
            (setq expnt (strcat "+" expnt))
    )
    (setq input (strcat temp "e" expnt))
    (print_expo (strcat temp "e" expnt) nil)
    (clear_var 'num)
)

;; Character occurence counter/
;; Character position determiner
(defun counter (val char var)
   (setq n 1 m 0)
   (while (/= (setq ch (substr val n 1)) "")
      (if (= ch char) 
          (progn
            (setq m (+ m 1))
            (if var (set (read (strcat var (itoa m))) n))
          )
      )
      (setq n (+ n 1))
   ) (eval m)
)

;; Calc-fraction function.
(defun com_fraction ()
   (if (not fval) 
      (if (and input (not (zerop (atof input)))
          (= (type (read input)) 'INT)) 
          (setq fval input m1 nil m2 nil)
      )
   )
   (cond
     ((and fval (not rdc))
       (setq lenf (strlen fval))
       (counter fval (chr 47) "m")
       (if (and num (/= num "dot") (<= lenf 21))
           (if (and (not (and (= num "/") 
                    (if (> lenf 0) (= (substr fval lenf) "/")))) 
                    (not (and m1 m2 (= num "/")))
               )
               (progn
                 (if (= num "-")
                     (setq fval (strcat 
                         (itoa (* -1 (atoi (substr fval 1 (- m1 1)))))
                         (substr fval m1))
                     )
                     (setq fval (strcat fval num))
                 )
               )
           )
       )
       (print_fraction fval)
     )
     (t (clear_var 'fract))
   )
)

;; Fractional input converter
;; (converts input in fraction format to real no.)
(defun fract_test ()
   (if fract 
     (cond
       ((setq rval (distof (eval_fract)))
        (digi_deter rval)
        (setq sval (rtos rval 2 prec))
       )
       (t (setq code "Invalid fraction")
          (print_error)
       )
     )
   )
   (clear_var (list 'm1 'm2 'fval 'fract))
)

;; Fraction input evaluator
(defun eval_fract ()
   (cond
     ((= (counter fval (chr 47) "k") 2)
      (setq fa (substr fval 1 (- k1 1)))
      (setq fb (substr fval (+ k1 1) (- k2 1)))
      (setq fc (substr fval (+ k2 1)))
      (if (= fc "") (setq fc "0"))
      (setq input 
        (strcat (itoa (+ (* (abs (atoi fa)) (atoi fc)) (atoi fb))) "/" fc)
      )
      (if (minusp (atoi fa)) (setq input (strcat "-" input)))
     )
     ((= (counter fval (chr 47) "k") 1)  
      (setq fa (substr fval 1 (- k1 1)))
      (setq fb (substr fval (+ k1 1)))
      (if (= fb "") (setq fb "0"))
      (setq input (strcat fa "/" fb))
     )
     (t (setq input fval))
   ) 
   (clear_var (list 'k1 'k2))
   (eval input)
)


;; Conversion of mixed fraction into normal form
(defun norm_fract ()
   (if fract
     (cond
       ((/= (counter (eval_fract) (chr 47) "m") 0)
        (cond
           ((distof (eval_fract))
            (setq a (atoi (substr (eval_fract) 1 (- m1 1))))
            (setq b (atoi (substr (eval_fract) (+ m1 1))))
            (setq c (gcd (abs a) b)) 
            (setq d (/ a c) e (/ b c))
            (if (= e 1) (print_fraction (itoa d))
              (print_fraction (strcat (itoa d) (chr 47) (itoa e)))
            )
           )
           (t (setq code "Invalid fraction")
              (print_error)
           )
        )
       )
       (t (print_fraction (eval_fract)))
     )
   )
   (clear_var (list 'm1 'a 'b 'c 'd 'e))
)

(defun ptor ()
   (if (not polar_pick)
      (progn
         (ptor_test)
         (if (not pr)
           (cond
             (input (setq pr (atof input)))
             (t (setq pr rval))
           )
         )
         (if ptr (print_value input nil))
         (setq ptr t input nil polar_pick t)
      )
   )
)

(defun rtop ()
   (if (not polar_pick)
      (progn
         (rtop_test)
         (if (not rp) (setq rp rval))
         (if rtp (print_value input nil))
         (setq rtp t input nil polar_pick t)
      )
   )
)

(defun ptor_test ()
   (if ptr
     (progn
       (if ansr 
         (setq exch 0.0)
         (setq exch (* pr (sin (/ (convert rval) (/ 180 pi)))))
       )
       (setq input (rtos (setq pr 
             (* pr (cos (/ (convert rval) (/ 180 pi))))) 2 100)
       )
     )
   )
)

(defun rtop_test ()
   (if rtp
      (progn
        (if ansr
          (setq exch 0.0)
          (setq exch (inverse (* (atan rval rp) (/ 180 pi))))
        )
        (setq input (rtos (setq rp 
              (sqrt (+ (expt rp 2.0) (expt rval 2.0)))) 2 100)
        )
      )
   )
)

(defun x-y ()
   (if (not exch) (setq exch 0.0))
   (if (and (not rtp) (not ptr)) (setq ansr rval))
   (num_eval exch)
   (setq exch rval input nil)
)

;; Sexagisemal input function.
(defun com_sexa ()
   (if (not ctr) (setq ctr 3))
   (cond
      ((= (rem ctr 3) 0) 
       (setq sex-val rval)
      )
      ((= (rem ctr 3) 1) 
       (setq sex-val (+ sex-val (/ rval 60.0)))
      )
      ((= (rem ctr 3) 2) 
       (setq sex-val (+ sex-val (/ rval 3600.0)))
      )
    )
    (setq ctr (+ ctr 1))
    (num_eval sex-val)
    (clear_var 'input)
)

;; Input list modifier.
(defun modify_list ()
    (cond
       ((or (and (= oper "*") (= opr "*"))
            (and (= oper "/") (= opr "*"))
        )
        (setq a (nth 3 (reverse numlst)))
        (setq b (* (nth 2 (reverse numlst)) 
                   (nth 1 (reverse numlst)))
        )
        (setq c (nth 0 (reverse numlst)))
        (setq numlst (list a b c))
        (setq opr oprr)
       )
       ((or (and (= oper "/") (= opr "/"))
            (and (= oper "*") (= opr "/"))
        )
        (setq a (nth 3 (reverse numlst)))
        (setq b (/ (nth 2 (reverse numlst)) 
                   (nth 1 (reverse numlst)))
        )
        (setq c (nth 0 (reverse numlst)))
        (setq numlst (list a b c))
        (setq opr oprr)
       )
       ((or (and (= oper "expt") (= opr "expt"))
            (and (= oper "invexpt") (= opr "expt"))
        )
        (setq a (nth 3 (reverse numlst)))
        (setq b (expt (nth 2 (reverse numlst)) 
                      (nth 1 (reverse numlst))) 
        )
        (setq c (nth 0 (reverse numlst)))
        (setq numlst (list a b c))
        (setq opr oprr)
       )
       ((or (and (= oper "invexpt") (= opr "invexpt"))
            (and (= oper "expt") (= opr "invexpt"))
        )
        (setq a (nth 3 (reverse numlst)))
        (setq b (expt (nth 2 (reverse numlst)) 
                      (/ 1.0 (nth 1 (reverse numlst))))
        )
        (setq c (nth 0 (reverse numlst)))
        (setq numlst (list a b c))
        (setq opr oprr)
       )
    )
    (setq val-a (nth 2 (reverse numlst)))
    (setq val-b (nth 1 (reverse numlst)))
    (setq val-d (nth 0 (reverse numlst)))
)

;; Operation interpreter.
(defun com_oper () 
    (defun equal_1 ()
       (if (and saver (not input)) (setq input saver saver nil))
       (if input (setq numlst (append numlst (list (distof input)))))
       (if input (setq m-lst nil))
       (if (not m-lst) (setq m-lst (list num)))
       (if (and (= num oper) (not input))
           (setq m-lst (append m-lst (list num)))
           (setq m-lst (list num))
       )
       (if (zerop (rem (length m-lst) 2))
           (progn
             (setq m-val rval m-oper (nth 0 m-lst))
             (set_tile "error" 
             (adjust "<Constant Calc. Mode activated>"))
           )
           (progn
             (clear_var (list 'm-val 'm-oper))
             (set_tile "error" "")
           )
       )
       (com_equal) 
       (if opr (setq oprr opr))
       (if input (if oper (setq opr oper)))
       (if num (setq oper num))
       (clear_var 'input)
    )
    (defun equal_2 ()
       (if (and saver (not input)) (setq input saver saver nil))
       (if (not input) 
           (if (and m-val m-oper) 
               (setq input (rtos (float ansr) 2 100))
           )
       )                                                   
       (if input (setq numlst (append numlst (list (distof input)))))
       (com_equal)
       (clear_var (list 'oper 'opr 'oprr 'numlst))
       (if ansr (setq numlst (append numlst (list ansr ))))
       (clear_var 'input)
    )
    (cond
      ((member num '( "+" "-" "/" "*" "expt" "invexpt")) 
       (if input (ptor_test))  
       (if input (rtop_test)) 
       (equal_1)
       (if (and (not ptr) (not rtp)) (setq exch ansr))
       (clear_var (list 'ptr 'rtp))
      )
      ((= num "=") 
       (ptor_test) (rtop_test) 
       (equal_2)
       (if (and (not ptr) (not rtp)) (setq exch ansr))
       (clear_var (list 'ptr 'rtp))
      )
    )
      
    (setq polar_pick nil);;;check whether polar key was hit 
                         ;;;more than once consecutively

;; Calculates the result on the specified operation.
(defun com_equal () 
    (cond
      ((and (= oper "+") (not m-oper)) 
       (if input (com_add))
      )
      ((and (= oper "-") (not m-oper)) 
       (if input (com_minus))
      )
      ((and (= oper "/") (not m-oper)) 
       (if input (com_divide))
      )
      ((and (= oper "*") (not m-oper)) 
       (if input (com_multiply))
      )
      ((and (= oper "expt") (not m-oper)) 
       (if input (com_expt))
      )
      ((and (= oper "invexpt") (not m-oper)) 
       (if input (com_invexpt))
      )
      (t (com_else))
    )
    (cond
      ((= ansr "error") (print_error))
      ((and (not (zerop ansr)) (> (abs ansr) 9.999999999999999e+99))
       (setq code "Infinity") (print_error)
      )
      (t (if (< (abs ansr) 1.0e-99) (setq ansr 0.0))
         (if (or (= num "=") (if input (eq num oper))) 
             (com_result ansr))
      )
    )
)

;; Takes care of the return value in case no 
;; previously entered arithmetic operation.
(defun com_else () 
   (if input
       (setq ansr (distof input))
       (setq ansr rval)
   )
   (if (and m-val m-oper (= num "="))
      (cond
        ((=  m-oper "expt") 
         (if (and (zerop ansr) (zerop m-val))
             (setq ansr "error" code "Divide by zero")
             (setq ansr (expt ansr m-val))
         )
        )
        ((=  m-oper "invexpt") 
         (if (or (and (zerop ansr) (zerop m-val)) (zerop m-val))
             (setq ansr "error" code "Divide by zero")
             (setq ansr (expt ansr (/ 1.0 m-val)))
         )
        )
        ((=  m-oper "/") 
         (if (zerop m-val)
             (setq ansr "error" code "Divide by zero")
             (setq ansr (/ ansr m-val))
         )
        )
        ((=  m-oper "*") (setq ansr (* ansr m-val)))
        ((=  m-oper "+") (setq ansr (+ ansr m-val)))
        ((=  m-oper "-") (setq ansr (- ansr m-val)))
      )
   )
)

 )

;; Exponential function.
(defun com_expt () 
   (modify_list)
   (if (and (zerop val-d) (zerop ansr))
       (setq ansr "error" code "Divide by zero")
       (progn
       (if (= opr "*")
          (cond
            ((= oprr "+") (setq val-c (- ansr (* val-a val-b)))
             (setq ansr (+ val-c (* (expt val-b val-d) val-a)))
            )
            ((= oprr "-") (setq val-c (+ ansr (* val-a val-b)))
             (setq ansr (- val-c (* (expt val-b val-d) val-a)))
            )
            ((or (= oprr "*") (= oprr "/") (not oprr))
             (setq ansr (* ansr (expt val-b (- val-d 1.0))))
            )
            ((= oprr "expt")
             (setq val-c (exp (/ (log (/ ansr val-b)) val-a)))
             (setq ansr (* (expt val-c val-a) (expt val-b val-d)))
            )
            ((= oprr "invexpt")
             (setq val-c (exp (/ (log (/ ansr val-b)) (/ 1.0 val-a))))
             (setq ansr (* (expt val-c (/ 1.0 val-a)) (expt val-b val-d)))
            )
          )
       )
       (if (= opr "/")
          (cond
            ((= oprr "+") (setq val-c (- ansr (/ val-a val-b)))
             (setq ansr (+ val-c (/ val-a (expt val-b val-d))))
            )
            ((= oprr "-") (setq val-c (+ ansr (/ val-a val-b)))
             (setq ansr (- val-c (/ val-a (expt val-b val-d))))
            )
            ((or (= oprr "*") (= oprr "/") (not oprr))
             (setq ansr (* ansr (expt val-b (- 1.0 val-d))))
            )
            ((= oprr "expt")
             (setq val-c (exp (/ (log (* ansr val-b)) val-a)))
             (setq ansr (/ (expt val-c val-a) (expt val-b val-d)))
            )
            ((= oprr "invexpt")
             (setq val-c (exp (/ (log (* ansr val-b)) (/ 1.0 val-a))))
             (setq ansr (/ (expt val-c (/ 1.0 val-a)) (expt val-b val-d)))
            )
          )
       )
       (if (= opr "+") (setq ansr (+ (- ansr val-b) (expt val-b val-d))))
       (if (= opr "-") (setq ansr (- (+ ansr val-b) (expt val-b val-d))))
       (if (or (not opr) (= opr "expt") (= opr "invexpt"))
           (setq ansr (expt ansr val-d))
       )
       )
   )
)

;; Standard deviation function
(defun deviation (output)
   (setq expo nil expnt nil)
   (if (not datalist)
       (setq num "error" code "Empty data entry")
       (cond
         ((= output "dtn")
          (num_eval (length datalist))
         )
         ((= output "dts")
          (num_eval (apply '+ datalist))
         )
         ((= output "ave")
          (num_eval (/ (apply '+ datalist) (length datalist)))
         )
         ((= output "sqr1")
          (num_eval (expt (apply '+ datalist) 2))
         )
         ((= output "sqr2")
          (num_eval (apply '+ (mapcar '* datalist datalist)))
         )
         ((= output "strd0")
          (if (zerop (apply '+ datalist))
              (setq num "error")
              (num_eval (sqrt (/ (- (apply '+ (mapcar '* datalist datalist)) 
                        (/ (expt (apply '+ datalist) 2) (length datalist))) 
                        (length datalist)))
              )
          )
         )
         ((= output "strd1")
          (if (or (zerop (apply '+ datalist)) (zerop (- (length datalist) 1)))
              (setq num "error")
              (num_eval (sqrt (/ (- (apply '+ (mapcar '* datalist datalist)) 
                        (/ (expt (apply '+ datalist) 2) (length datalist))) 
                        (- (length datalist) 1)))
              )
          )
         )
       )
   )
   (clear_var 'input)
)

;; Significant digits determiner.
(defun digi_deter (digit)
   (setq val1 (chr 48) jn 15)
   (setq dm (strlen (rtos (abs (float digit)) 2 0)))
   (while (= val1 (chr 48))
      (setq val1 (substr (rtos (abs 
                 (float digit)) 2 (- 14 dm)) jn 1)
      )
      (setq jn (- jn 1))
   )
   (setq prec (- jn dm))
   (setq prec1 prec)
   (cond
      (random (setq prec 3)) 
      (t (if fxn (setq prec fxn)))
   )
)

;; Results evaluator function.
(defun com_result (val)  
    (if (and (<= (abs val) 1e-10) (not (zerop val)))
        (cond
          (fxn (print_value (rtos (float val) 1 fxn) val))
          (scn (print_value (rtos (float val) 1 scn) val))
          (t (print_value (rtos (float val) 1 15) val))
        )
        (if (>= (abs val) 1e+11) 
            (cond
              (fxn (print_value (rtos (float val) 1 fxn) val))
              (scn (print_value (rtos (float val) 1 scn) val))
              (t (print_value (rtos (float val) 1 15) val))
            )
            (progn
              (cond
                ((and (not random) scn) 
                 (print_value (rtos (float val) 1 scn) val)
                )
                (t (digi_deter val)
                   (setq new (rtos (float val) 2 prec))
                   (if (> (strlen new) max_digit)
                       (setq new (rtos (float val) 2 prec1))
                   )
                   (if (= (substr new 1 1) ".")
                       (setq new (strcat "0" new))
                   )
                   (if (= (substr new 1 2) "-.")
                       (setq new (strcat "-0" (substr new 2)))
                   )
                   (print_value new val)
                )
              )
            )
        )
    )
)

;; Function evaluator.
(defun num_eval (num-val) 
    (setq rmv (float num-val))
    (setq dmv (rtos (float num-val) 2 100))
    (if (and (<= (abs num-val) 1e-10) (not (zerop num-val)))
        (setq num (rtos (float num-val) 1 15))
        (if (>= (abs num-val) 1e+11) 
            (cond
              (fxn (setq num (rtos (float num-val) 1 fxn)))
              (scn (setq num (rtos (float num-val) 1 scn)))
              (t (setq num (rtos (float num-val) 1 15)))
            )
            (progn
              (cond
                ((and (not random) scn) 
                 (setq num (rtos (float num-val) 1 scn))
                )
                (t (digi_deter num-val)
                   (setq num (rtos (float num-val) 2 prec))
                   (if (> (strlen num) max_digit)
                       (setq num (rtos (float num-val) 2 prec1))
                   )
                )
              )
           )
        )
    )
    (if (= (substr num 1 1) ".")
        (setq num (strcat "0" num))
    )
    (if (= (substr num 1 2) "-.")
        (setq num (strcat "-0" (substr num 2)))
    )
    (setq funct t random nil)
    (cond
      ((and (not (zerop rmv)) (> (abs rmv) 9.999999999999999e+99))
       (setq num "error" code "Infinity")
      )
      (t (if (< (abs rmv) 1.0e-99) (setq num "0"))
         (eval num)
      )
    )
)

;;Memory entry evaluator
(defun evaluate_mem ()
   (setq m 97 ini_lst nil)
   (while (<= m 122)
      (if (eval (read (strcat "mem" (chr m))))
          (if (not (zerop (eval (read (strcat "mem" (chr m))))))
              (setq ini_lst (append ini_lst 
                    (list (read (strcat "mem" (chr m))))))
          )
      ) 
      (setq m (+ m 1))
   )
   ini_lst
)

;;Memory save function
(defun memory_sav ()
  (if (or (not (eq orig_mem (evaluate_mem))) ;; evaluate whether memory
          (not (eq (mapcar 'eval orig_mem)   ;; entry was changed.
               (mapcar 'eval (evaluate_mem))))  ;; If so, rewrite file.
      )
     (progn
       (setq fn (findfile "com.lsp") m 97)
       (setq f (open (strcat (substr fn 1 (- (strlen fn) 3)) "mem") "w"))
       (write-line ";;;Calculator Memory ini file" f)
       (write-line ";;;Please do not alter or erase" f)
       (while (<= m 122)
         (if (eval (read (strcat "mem" (chr m))))
             (if (not (zerop (eval (read (strcat "mem" (chr m))))))
                 (progn
                     (setq vv (eval (read (strcat "mem" (chr m)))))
			   (if (minusp vv) 
                         (write-line (strcat "(setq mem" (chr m) "   " (rtos vv 2 15) ")") f)
                         (write-line (strcat "(setq mem" (chr m) "  0" (rtos vv 2 15) ")") f)
                     )
		     )
             )
         ) (setq m (+ m 1))
       ) (setq f (close f))
     )
  )
)
